cmake_minimum_required(VERSION 3.28)
project(rook_nav2_experiment LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(action_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)

execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare.py
  ${CMAKE_CURRENT_BINARY_DIR}/upstream COMMAND_ERROR_IS_FATAL ANY)

# Build the identified BT implementation separately from the ROS-installed one.
include(FetchContent)
set(BTCPP_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BTCPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BTCPP_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BTCPP_GROOT_INTERFACE OFF CACHE BOOL "" FORCE)
set(BTCPP_SQLITE_LOGGING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(btcpp
  EXCLUDE_FROM_ALL
  URL https://codeload.github.com/BehaviorTree/BehaviorTree.CPP/tar.gz/3ff6a32ba0497a08519c77a1436e3b81eff1bcd6
  URL_HASH SHA256=84486d12ce7249a611726dd4fbf4a43a9b2f3abf0b00448b6d910a3a54d9a281)
set(CMAKE_DISABLE_FIND_PACKAGE_ament_cmake TRUE)
FetchContent_MakeAvailable(btcpp)
unset(CMAKE_DISABLE_FIND_PACKAGE_ament_cmake)

enable_testing()
add_test(NAME nav2_property COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_property.py)
add_test(NAME nav2_adapter COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_adapter.py)
foreach(variant old fixed)
  add_executable(nav2_${variant} experiment.cpp)
  target_compile_features(nav2_${variant} PRIVATE cxx_std_20)
  target_compile_options(nav2_${variant} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(nav2_${variant} PRIVATE UPSTREAM_HEADER="${variant}.hpp" VARIANT="${variant}")
  target_include_directories(nav2_${variant} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/upstream)
  ament_target_dependencies(nav2_${variant} rclcpp rclcpp_action action_msgs nav2_msgs)
  target_link_libraries(nav2_${variant} behaviortree_cpp Threads::Threads)
  add_test(NAME control_${variant} COMMAND nav2_${variant})
  set_tests_properties(control_${variant} PROPERTIES ENVIRONMENT "RMW_IMPLEMENTATION=rmw_fastrtps_cpp")
endforeach()

foreach(variant old fixed noop always-cancel)
  if(variant STREQUAL "old")
    set(source old)
    set(id_offset 0)
  else()
    set(source fixed)
    set(id_offset 1)
  endif()
  if(variant STREQUAL "old" OR variant STREQUAL "fixed")
    set(mutation none)
  else()
    set(mutation ${variant})
  endif()
  add_executable(nav2_component_${variant} component.cpp)
  target_compile_features(nav2_component_${variant} PRIVATE cxx_std_20)
  target_compile_options(nav2_component_${variant} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(nav2_component_${variant} PRIVATE
    UPSTREAM_HEADER="${source}.hpp" MUTATION="${mutation}" COMPONENT_VARIANT="${variant}" GOAL_ID_OFFSET=${id_offset})
  target_include_directories(nav2_component_${variant} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/upstream)
  ament_target_dependencies(nav2_component_${variant} rclcpp rclcpp_action action_msgs nav2_msgs)
  target_link_libraries(nav2_component_${variant} behaviortree_cpp Threads::Threads nlohmann_json::nlohmann_json)
endforeach()
ament_package()
